set(GL_TEST_MAP "${GL_HOST_DIR}/libs/ddnet_physics/tests/Aip-Gores.map")
foreach(game ddnet unique)
    set(physics ${game}_physics)
    if(NOT TARGET ${physics})
        continue()
    endif()
    add_executable(grenade_lineups_parity_${game} flight_parity.cpp "${GL_SOURCE_DIR}/gl_table.cpp" "${GL_SOURCE_DIR}/gl_fast.cpp")
    target_include_directories(grenade_lineups_parity_${game} PRIVATE "${GL_SOURCE_DIR}")
    target_link_libraries(grenade_lineups_parity_${game} PRIVATE ${physics} Threads::Threads)
    if(NOT MSVC)
        target_compile_options(grenade_lineups_parity_${game} PRIVATE -O2 -ffp-contract=off)
    endif()
    add_test(NAME grenade_lineups_parity_${game} COMMAND grenade_lineups_parity_${game} "${GL_TEST_MAP}")
endforeach()
